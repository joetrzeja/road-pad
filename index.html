const drawBtn = document.getElementById('toggle-draw'); drawBtn.addEventListener('click', async () => { isDrawingMode = !isDrawingMode; canvas.style.pointerEvents = isDrawingMode ? 'auto' : 'none'; drawBtn.innerText = isDrawingMode ? '✏️ Draw: ON' : '✏️ Draw: OFF'; drawBtn.classList.toggle('active', isDrawingMode); if (isDrawingMode) { deselectAllCars(); // 1. Lock map dragging and zooming gestures map.dragging.disable(); map.touchZoom.disable(); map.doubleClickZoom.disable(); map.boxZoom.disable(); map.keyboard.disable(); // 2. Lock Map Rotation (Freeze current bearing and disable rotation touch handlers) if (map.touchRotate) { map.touchRotate.disable(); } // Lock the rotate slider so the user cannot manually spin the map while drawing document.getElementById('rotate-slider').disabled = true; // 3. Disable car dragging carMarkers.forEach(c => c.dragging.disable()); // 4. Lock physical device screen orientation (Browser/PWA API) if (screen.orientation && screen.orientation.lock) { try { await screen.orientation.lock(screen.orientation.type); } catch (err) { console.warn('Device orientation lock not supported in standard browser mode:', err); } } } else { // Re-enable Map Dragging and Zooming map.dragging.enable(); map.touchZoom.enable(); map.doubleClickZoom.enable(); map.boxZoom.enable(); map.keyboard.enable(); // Re-enable Map Rotation if (map.touchRotate) { map.touchRotate.enable(); } document.getElementById('rotate-slider').disabled = false; // Re-enable Car Dragging for unpinned cars carMarkers.forEach(c => { if (!c.isPinned) c.dragging.enable(); }); // Unlock physical device screen orientation if (screen.orientation && screen.orientation.unlock) { try { screen.orientation.unlock(); } catch (err) { console.warn('Device orientation unlock failed:', err); } } } });
